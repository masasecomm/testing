Scans from an open Google tab in the background; you can use any tab. Sheets sync runs in the background.
  1. Create a new Google Sheet.
  2. Open Extensions → Apps Script.
  3. Replace the default code with the script below, then click Save.
  4. Click Deploy → New deployment → Web app.
  5. Set Execute as: Me, Who has access: Anyone.
  6. Click Deploy, copy the /exec URL, paste it above and click Save URL.
function doPost(e) {
  const data = JSON.parse(e.postData.contents);
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const SHEETS = ['All Captured Queries','New Entries','Rising Queries','Dying Queries','Published Queries'];
  // Ensure all 5 sheets exist
  SHEETS.forEach(name => { if (!ss.getSheetByName(name)) ss.insertSheet(name); });

  function writeQuery(sheetName, rows, meta) {
    const s = ss.getSheetByName(sheetName);
    s.clear();
    let r = 1;
    if (meta) { s.getRange(r,1,1,2).setValues([['Exported (Local Time)', meta]]); r++; }
    s.getRange(r,1,1,3).setValues([['Rank','Search Query','Times Seen']]); r++;
    if (rows && rows.length) {
      const vals = rows.map(x => [x.rank, x.query, x.count]);
      s.getRange(r,1,vals.length,3).setValues(vals);
    }
  }

  if (data.action === 'reset') {
    SHEETS.forEach(n => ss.getSheetByName(n).clear());
    return ContentService.createTextOutput(JSON.stringify({ ok:true, reset:true }))
      .setMimeType(ContentService.MimeType.JSON);
  }

  if (data.action === 'publish') {
    const s = ss.getSheetByName('Published Queries');
    // Ensure headers in A1:B1
    s.getRange(1,1,1,2).setValues([['Published Query','Published At (Local Time)']]);
    // Find first empty row in column A
    const colA = s.getRange('A:A').getValues();
    let nextRow = 0;
    for (let i = 0; i < colA.length; i++) {
      if (colA[i][0] === '' || colA[i][0] === null) { nextRow = i + 1; break; }
    }
    if (nextRow === 0) nextRow = colA.length + 1;
    // Write query + timestamp so Make.com "Watch Changes" detects a new row
    s.getRange(nextRow, 1, 1, 2).setValues([[data.query, data.publishedAtLocalTime || '']]);
    // Force Sheets to commit the write so external integrations detect the new row
    SpreadsheetApp.flush();
    return ContentService.createTextOutput(JSON.stringify({ ok:true, published:data.query, row:nextRow }))
      .setMimeType(ContentService.MimeType.JSON);
  }

  // Default: bucketSync — refresh the 4 query sheets, leave Published untouched
  writeQuery('All Captured Queries', data.all || [], data.exportedAtLocalTime);
  writeQuery('New Entries', data.newEntries || []);
  writeQuery('Rising Queries', data.rising || []);
  writeQuery('Dying Queries', data.dying || []);
  // Ensure Published header exists but never wipe it
  const pub = ss.getSheetByName('Published Queries');
  pub.getRange(1,1,1,2).setValues([['Published Query','Published At (Local Time)']]);

  return ContentService.createTextOutput(JSON.stringify({ ok:true }))
    .setMimeType(ContentService.MimeType.JSON);
}
