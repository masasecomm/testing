1. Create a new Google Sheet.
  2. Open Extensions → Apps Script.
  3. Replace the default code with the script below, then click Save.
  4. Click Deploy → New deployment → Web app.
  5. Set Execute as: Me, Who has access: Anyone.
  6. Click Deploy, copy the /exec URL, paste it above and click Save URL.
function doPost(e) {
  const data = JSON.parse(e.postData.contents);
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const SHEETS = ['All Captured Queries','New Entries','Rising Queries','Dying Queries','TimeStamp'];
  // Ensure all sheets exist
  SHEETS.forEach(name => { if (!ss.getSheetByName(name)) ss.insertSheet(name); });

  function writeQuery(sheetName, rows, meta) {
    const s = ss.getSheetByName(sheetName);
    s.clear();
    let r = 1;
    if (meta) { s.getRange(r,1,1,2).setValues([['Exported (CAT)', meta]]); r++; }
    s.getRange(r,1,1,3).setValues([['Rank','Search Query','Times Seen']]); r++;
    if (rows && rows.length) {
      const vals = rows.map(x => [x.rank, x.query, x.count]);
      s.getRange(r,1,vals.length,3).setValues(vals);
    }
  }

  function writeTimestamps(rows) {
    const sheet = ss.getSheetByName('TimeStamp');
    sheet.clear();
    sheet.getRange(1,1,1,3).setValues([['Captured At (CAT)','Search Query','Scan']]);
    if (!rows || !rows.length) return;
    const values = rows.map(row => [
      Utilities.formatDate(new Date(row.capturedAt), 'Africa/Harare', 'yyyy-MM-dd HH:mm:ss'),
      row.query,
      row.scan,
    ]);
    sheet.getRange(2,1,values.length,3).setValues(values);
  }

  if (data.action === 'reset') {
    SHEETS.forEach(n => ss.getSheetByName(n).clear());
    return ContentService.createTextOutput(JSON.stringify({ ok:true, reset:true }))
      .setMimeType(ContentService.MimeType.JSON);
  }

  // Default: bucketSync — refresh the query sheets and timestamp log
  writeQuery('All Captured Queries', data.all || [], data.exportedAtCAT);
  writeQuery('New Entries', data.newEntries || []);
  writeQuery('Rising Queries', data.rising || []);
  writeQuery('Dying Queries', data.dying || []);
  writeTimestamps(data.timestampEntries || []);
  return ContentService.createTextOutput(JSON.stringify({ ok:true }))
    .setMimeType(ContentService.MimeType.JSON);
}
