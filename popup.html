function doPost(e) {
  const data = JSON.parse(e.postData.contents);
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const SHEETS = ['All Captured Queries','New Entries','Rising Queries','Dying Queries'];
  // Ensure all 4 sheets exist
  SHEETS.forEach(name => { if (!ss.getSheetByName(name)) ss.insertSheet(name); });

  function writeQuery(sheetName, rows, meta) {
    const s = ss.getSheetByName(sheetName);
    s.clear();
    let r = 1;
    if (meta) { s.getRange(r,1,1,2).setValues([['Exported (CAT)', meta]]); r++; }
    s.getRange(r,1,1,3).setValues([['Rank','Search Query','Times Seen']]); r++;
    if (rows && rows.length) {
      const vals = rows.map(x => [x.rank, x.query, x.count]);
      s.getRange(r,1,vals.length,3).setValues(vals);
    }
  }

  if (data.action === 'reset') {
    SHEETS.forEach(n => ss.getSheetByName(n).clear());
    return ContentService.createTextOutput(JSON.stringify({ ok:true, reset:true }))
      .setMimeType(ContentService.MimeType.JSON);
  }

  // Default: bucketSync — refresh the 4 query sheets
  writeQuery('All Captured Queries', data.all || [], data.exportedAtCAT);
  writeQuery('New Entries', data.newEntries || []);
  writeQuery('Rising Queries', data.rising || []);
  writeQuery('Dying Queries', data.dying || []);
  return ContentService.createTextOutput(JSON.stringify({ ok:true }))
    .setMimeType(ContentService.MimeType.JSON);
}
